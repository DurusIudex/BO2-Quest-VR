cmake_minimum_required(VERSION 3.22)
project(BO2QuestVR LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/vendor/dobby" dobby)
add_library(openxr_loader SHARED IMPORTED)
set_target_properties(openxr_loader PROPERTIES IMPORTED_LOCATION "${ROOT}/toolchain/android/openxr-loader/jni/arm64-v8a/libopenxr_loader.so")
add_library(questvr SHARED quest_runtime.cpp quest_input.cpp controller_bindings.cpp motion_gestures.cpp guest_camera.cpp guest_bridge.S relay_transport.cpp)
target_include_directories(questvr PRIVATE "${ROOT}/vendor/openxr/include")
target_compile_options(questvr PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_link_libraries(questvr PRIVATE dobby openxr_loader android log dl)

# Explicit test targets only; neither is packaged in the APK.
add_library(relay_test_host SHARED EXCLUDE_FROM_ALL relay_test_host.cpp)
set_target_properties(relay_test_host PROPERTIES OUTPUT_NAME t6host LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/relay-test")
add_executable(relay_transport_test EXCLUDE_FROM_ALL relay_transport_test.cpp relay_transport.cpp)
target_compile_options(relay_transport_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(relay_transport_test PRIVATE dobby log dl)
